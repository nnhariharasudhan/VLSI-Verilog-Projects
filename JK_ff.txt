VERILOG
module jk_ff (
  input  j,k,
  input  clk,
  output reg q
);
  always @(posedge clk)
    begin 
      case ({j,k})
      2'b00: q <= q;
      2'b01: q <= 1'b0;
      2'b10: q <= 1'b1;
      2'b11: q <= ~q;
      endcase
      end
endmodule

module jk_fftb;
  reg j,k;
  reg clk;
  wire q;
  
  jk_ff dut(.j(j),
            .k(k),
            .clk(clk),
            .q(q)
           );
  always #5 clk = ~clk;
  initial begin 
  clk = 0;
 $display("Time\tJ K\tQ");
    $monitor("%0t\t%b %b\t%b", $time, j, k, q);

endmodule
